% Flush the previous problem BEFORE updating its metadata and page marks.
\newcommand{\problemheader}[2]{%
    \clearpage
    \setproblem{#1}{#2}%
    \markboth{Problem #1}{Problem #1}%
    \phantomsection
    \addcontentsline{toc}{section}{#1. #2}%
    \begin{center}
        {\Large\bfseries Problem #1\par}
        \vspace{0.4em}
        {\LARGE\bfseries #2\par}
    \end{center}
}
