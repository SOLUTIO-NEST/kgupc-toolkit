% Flush the previous problem BEFORE updating its metadata and page marks.
\newcommand{\problemheader}[2]{%
    \clearpage
    \setproblem{#1}{#2}%
    \markboth{Problem #1}{Problem #1}%
    \phantomsection
    \addcontentsline{toc}{section}{\protect\numberline{#1.}#2}%
    \begin{center}
        {\Large\bfseries Problem #1\par}
        \vspace{0.4em}
        {\LARGE\bfseries #2\par}
    \end{center}
}

% Limits accept numbers in seconds and MB; empty values remain hidden.

            시간 제한: #1초%
            \ifstrempty{#2}{}{\quad\textbar\quad 메모리 제한: #2MB}%
        \end{center}
    }%
}
